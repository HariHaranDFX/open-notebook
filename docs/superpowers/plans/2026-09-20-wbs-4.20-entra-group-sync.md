> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship WBS 4.20 — an admin can link a Microsoft Entra security group into Open Notebook and its membership stays reflected in the ACL automatically.

**Architecture:** New Graph app-only client (`api/graph_client.py`) → new sync command (`commands/entra_group_sync.py`) → three new admin routes on `groups` router → in-process `asyncio` periodic trigger from the FastAPI lifespan. Schema unchanged (migration 28 reserved the fields). Frontend adds a "Link Entra group" dialog + "Synced from Entra" badge on existing groups page.

**Tech Stack:** FastAPI, `httpx.AsyncClient`, `surreal-commands`, SurrealDB, Next.js + shadcn UI, TanStack Query.

## Global Constraints

- **Design source of truth:** [docs/superpowers/specs/2026-09-20-wbs-4.20-entra-group-sync-design.md](../specs/2026-09-20-wbs-4.20-entra-group-sync-design.md). Every task implements one section of that spec.
- **No new Python dependency.** `httpx` (already used by `api/auth/entra.py`) covers Graph calls; `asyncio` covers the loop.
- **No schema change** — `user_group.source` and `user_group.entra_group_oid` already exist (migration 28).
- **Opt-in feature.** `ENTRA_GROUP_SYNC_ENABLED=false` by default. When disabled, no lifespan task starts, and the three new routes still register but the sync command no-ops (so admin UI degrades cleanly).
- **Admin-only writes.** All new mutating routes use `require_admin(request)`. Search endpoint returns 403 for non-admins.
- **i18n:** every new UI string exists in **all 16 locales** (parity + placeholder tests will fail otherwise).
- **Logging:** never log emails from Graph responses. Counts only.
- **License:** no GPL/AGPL packages. `httpx` is BSD, already in the tree.
- **Model A tenancy:** grants never cross deployments. `client_id` on `user` is unchanged.

```python
"""Microsoft Graph client — app-only auth + shared HTTP concerns.

Uses client-credentials flow with the existing Entra app registration
(ENTRA_TENANT_ID / ENTRA_CLIENT_ID / ENTRA_CLIENT_SECRET). Requires the
`GroupMember.Read.All` application permission with admin consent.

Nothing here is imported unless the caller (sync command or admin route)
runs; a deployment that leaves ENTRA_GROUP_SYNC_ENABLED unset never touches
Graph.
"""

from __future__ import annotations

import os
import time
from typing import Optional, Tuple

import httpx

_TOKEN_CACHE: Optional[Tuple[str, float]] = None


def _reset_token_cache() -> None:
    """Test-only helper — resets the module-level cache."""
    global _TOKEN_CACHE
    _TOKEN_CACHE = None


def _entra_env() -> tuple[str, str, str]:
    tenant = os.environ.get("ENTRA_TENANT_ID", "").strip()
    client_id = os.environ.get("ENTRA_CLIENT_ID", "").strip()
    client_secret = os.environ.get("ENTRA_CLIENT_SECRET", "").strip()
    if not (tenant and client_id and client_secret):
        raise RuntimeError(
            "Entra group sync requires ENTRA_TENANT_ID, ENTRA_CLIENT_ID, "
            "ENTRA_CLIENT_SECRET"
        )
    return tenant, client_id, client_secret


async def acquire_graph_token(client: httpx.AsyncClient) -> str:
    """Acquire an app-only Graph token, cached in-memory until 60s before expiry."""
    global _TOKEN_CACHE
    now = time.monotonic()
    if _TOKEN_CACHE is not None:
        token, expires_at = _TOKEN_CACHE
        if now < expires_at:
            return token
    tenant, client_id, client_secret = _entra_env()
    resp = await client.post(
        f"https://login.microsoftonline.com/{tenant}/oauth2/v2.0/token",
        data={
            "client_id": client_id,
            "client_secret": client_secret,
            "grant_type": "client_credentials",
            "scope": "https://graph.microsoft.com/.default",
        },
        timeout=10,
    )
    if resp.status_code != 200:
        raise RuntimeError(
            f"Graph token request failed: {resp.status_code} {resp.text}"
        )
    payload = resp.json()
    token = payload["access_token"]
    ttl = int(payload.get("expires_in", 3600))
    _TOKEN_CACHE = (token, now + max(ttl - 60, 30))
    return token
```

```python
_GRAPH_BASE = "https://graph.microsoft.com/v1.0"


async def _authed_get(client: httpx.AsyncClient, url: str, **kwargs) -> httpx.Response:
    token = await acquire_graph_token(client)
    headers = kwargs.pop("headers", {}) or {}
    headers["Authorization"] = f"Bearer {token}"
    resp = await client.get(url, headers=headers, timeout=15, **kwargs)
    return resp


async def search_groups(query: str, *, limit: int = 25) -> list[dict]:
    """Typeahead against /groups?$search — returns top `limit` matches, mapped."""
    async with httpx.AsyncClient() as client:
        resp = await _authed_get(
            client,
            f"{_GRAPH_BASE}/groups",
            params={
                "$search": f'"displayName:{query}"',
                "$top": str(limit),
                "$select": "id,displayName,description",
            },
            headers={"ConsistencyLevel": "eventual"},
        )
    if resp.status_code != 200:
        raise RuntimeError(
            f"Graph search failed: {resp.status_code} {resp.text}"
        )
    return [
        {
            "entra_group_oid": row.get("id", ""),
            "display_name": row.get("displayName", ""),
            "description": row.get("description"),
        }
        for row in resp.json().get("value", [])
    ]


async def get_group(entra_group_oid: str) -> dict:
    async with httpx.AsyncClient() as client:
        resp = await _authed_get(
            client,
            f"{_GRAPH_BASE}/groups/{entra_group_oid}",
            params={"$select": "id,displayName,description"},
        )
    if resp.status_code != 200:
        raise RuntimeError(
            f"Graph get_group failed: {resp.status_code} {resp.text}"
        )
    row = resp.json()
    return {
        "entra_group_oid": row.get("id", ""),
        "display_name": row.get("displayName", ""),
        "description": row.get("description"),
    }


async def list_group_member_oids(entra_group_oid: str) -> list[str]:
    """Walks paging; returns only direct #microsoft.graph.user members."""
    oids: list[str] = []
    url = f"{_GRAPH_BASE}/groups/{entra_group_oid}/members"
    async with httpx.AsyncClient() as client:
        while url:
            resp = await _authed_get(
                client, url, params={"$select": "id,mail,userPrincipalName"}
                if "$skiptoken" not in url
                else None,
            )
            if resp.status_code != 200:
                raise RuntimeError(
                    f"Graph list members failed: {resp.status_code} {resp.text}"
                )
            payload = resp.json()
            for row in payload.get("value", []):
                if row.get("@odata.type") == "#microsoft.graph.user":
                    oid = row.get("id")
                    if oid:
                        oids.append(oid)
            url = payload.get("@odata.nextLink", "")
    return oids
```

```python
"""Sync Entra ID group membership into local user_group_member.

Runs as a surreal-commands @command so it can be scheduled by the API
lifespan loop OR triggered manually via POST /api/groups/entra/sync.

Idempotent: each run diffs Entra membership against local membership and
applies only the delta. Members present in Entra but unknown locally
(no user row with matching entra_oid) are counted and skipped — they
auto-attach at first login. No new schema (uses fields reserved in
migration 28).
"""

from __future__ import annotations

from typing import Any, Dict, Optional

from loguru import logger
from pydantic import BaseModel
from surreal_commands import CommandInput, CommandOutput, command

from api.graph_client import get_group, list_group_member_oids
from open_notebook.database.repository import ensure_record_id, repo_query


class SyncEntraGroupsInput(CommandInput):
    group_id: Optional[str] = None


@command("sync_entra_groups", app="open_notebook", max_attempts=1)
async def sync_entra_groups(input_data: SyncEntraGroupsInput) -> CommandOutput:
    if input_data.group_id:
        rows = await repo_query(
            "SELECT id, entra_group_oid, name FROM $gid WHERE source = 'entra'",
            {"gid": ensure_record_id(input_data.group_id)},
        ) or []
    else:
        rows = await repo_query(
            "SELECT id, entra_group_oid, name FROM user_group WHERE source = 'entra'"
        ) or []

    totals: Dict[str, int] = {
        "groups_synced": 0,
        "members_added": 0,
        "members_removed": 0,
        "members_skipped_unknown": 0,
    }

    for row in rows:
        gid = str(row.get("id", ""))
        entra_oid = row.get("entra_group_oid")
        if not gid or not entra_oid:
            continue
        try:
            snapshot = await get_group(entra_oid)
            member_oids = await list_group_member_oids(entra_oid)
        except Exception as exc:
            logger.warning(
                f"entra_group_sync gid={gid} failed to fetch: {exc.__class__.__name__}"
            )
            continue

        # Refresh the name/description snapshot alongside membership.
        await repo_query(
            "UPDATE $gid SET name = $name, description = $desc",
            {
                "gid": ensure_record_id(gid),
                "name": snapshot.get("display_name") or row.get("name") or "",
                "desc": snapshot.get("description"),
            },
        )

        added, removed, skipped = await _reconcile_members(gid, member_oids)
        totals["groups_synced"] += 1
        totals["members_added"] += added
        totals["members_removed"] += removed
        totals["members_skipped_unknown"] += skipped
        logger.info(
            f"entra_group_sync gid={gid} added={added} removed={removed} "
            f"skipped_unknown={skipped}"
        )

    return CommandOutput(
        success=True,
        result=totals,
    )


async def _reconcile_members(gid: str, entra_oids: list[str]) -> tuple[int, int, int]:
    """Diff Entra membership against local; apply INSERT/DELETE deltas."""
    entra_set = set(entra_oids)
    known: list[Dict[str, Any]] = (
        await repo_query(
            "SELECT id, entra_oid FROM user WHERE entra_oid IN $oids",
            {"oids": list(entra_set)},
        )
        if entra_set
        else []
    ) or []
    entra_to_local = {row["entra_oid"]: str(row["id"]) for row in known}
    skipped_unknown = len(entra_set) - len(entra_to_local)
    desired_user_ids = set(entra_to_local.values())

    existing = await repo_query(
        "SELECT user_id FROM user_group_member WHERE group_id = $gid",
        {"gid": ensure_record_id(gid)},
    ) or []
    existing_user_ids = {str(row["user_id"]) for row in existing}

    to_add = desired_user_ids - existing_user_ids
    to_remove = existing_user_ids - desired_user_ids

    for uid in to_add:
        await repo_query(
            "CREATE user_group_member SET group_id = $gid, user_id = $uid",
            {"gid": ensure_record_id(gid), "uid": ensure_record_id(uid)},
        )
    for uid in to_remove:
        await repo_query(
            "DELETE user_group_member WHERE group_id = $gid AND user_id = $uid",
            {"gid": ensure_record_id(gid), "uid": ensure_record_id(uid)},
        )
    return len(to_add), len(to_remove), skipped_unknown
```
